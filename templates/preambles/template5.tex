% \symonrunningheads{math}.

\newcommand{\symonstatementmark}[2]{\extramarks{#1~#2}{#1~#2}}

  \symonstatementmark{#1}{\thechapter-\number\numexpr\value{theorem}+1\relax}}
\newcommand{\symonmathversomark}{%
  \ifblank{\firstleftxmark}{}{[\firstleftxmark}}
\newcommand{\symonmathrectomark}{%
  \ifblank{\firstleftxmark}{}{\firstleftxmark]}}

\fancypagestyle{symonmain}{%
  \fancyhf{}%
  \fancyhead[LE]{\thepage}%
  \fancyhead[CE]{\small\MakeUppercase{\leftmark}}%
  \fancyhead[RE]{\small
    \ifdefstring{\symonrunningheadmode}{math}
      {\symonmathversomark}
      {[CHAP.\ \thechapter]}}%
  \fancyhead[LO]{\small
    \ifdefstring{\symonrunningheadmode}{math}
      {\symonmathrectomark}
      {[\thesection]}}%
  \fancyhead[CO]{\small
    \ifdefstring{\symonrunningheadmode}{math}
      {\MakeUppercase{\leftmark}}
      {\MakeUppercase{\rightmark}}}%
  \fancyhead[RO]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
